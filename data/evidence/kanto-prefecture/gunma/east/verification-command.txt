import sys,json,hashlib,subprocess,datetime
from pathlib import Path
from shapely.geometry import shape
from shapely.ops import unary_union
sys.path.insert(0,'tools/national');from research_common import rounded_coordinate,decode_tile,combine_tiles
E=Path('data/evidence/kanto-prefecture/gunma/east');N=Path('data/national/kanto-prefecture/gunma/east');hashes={}
def digest(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
for p in list(E.glob('*.json'))+list(N.glob('*.json')):json.loads(p.read_text());hashes[str(p)]=digest(p)
for row in json.loads((E/'retrievals.json').read_text()):
 m=row['retrieval'];assert m['status'] in ['acquired','reused'];assert digest(m['rawPath'])==m['sha256']
for row in json.loads((E/'distribution-originals.json').read_text())['reusedRecords']:
 m=row['retrieval'];assert digest(m['rawPath'])==m['sha256']
for name in ['kiryu','maebashi','isesaki','midori']:
 g=json.loads((E/f'{name}-geometry.json').read_text());tiles=[]
 for s in g['sourceTiles']:
  p=Path(s['path']);assert digest(p)==s['sha256'];hashes[str(p)]=digest(p);d=decode_tile(p.read_bytes());features=[]
  for key in ['BldA','RdCL','RdEdg','WA','WL']:
   if key in d:features.extend(dict(f,layer=key) for f in d[key]['features'] if f['geometry']['type'] in ['Polygon','MultiPolygon','LineString','MultiLineString'])
  tiles.append(dict(tile=s['tile'],extent=4096,features=features))
 c=combine_tiles(tiles)
 for f in g['comparedFeatures']:assert c['features'][f['combinedIndex']]['geometry']==f['geometry']
 if g['selectedCombinedIndices']:
  union=unary_union([shape(c['features'][i]['geometry']) for i in g['selectedCombinedIndices']]);assert union.equals(shape(g['unionGeometry']));r=rounded_coordinate(g['unionGeometry'],g['baseTile'],g['pointXY']);assert r==g['coordinateCheck'];assert r['mechanicalChecksPassed'];assert not g['outerBoundaryTouched'];print(name,'source binding/union/rounded containment PASS',r['coordinate'])
 else:print(name,'comparison-only source binding PASS; no adopted coordinate')
loc=json.loads((N/'locations-proposed.json').read_text());assert len(loc['updates'])==3
for u in loc['updates']:assert digest(u['evidence']['snapshotPath'])==u['evidence']['snapshotSHA256'];assert u['evidence']['checkedAt']=='2026-11-06';assert u['coordinate']==json.loads(Path(u['evidence']['snapshotPath']).read_text())['coordinateCheck']['coordinate']
out=json.loads((N/'outcomes.json').read_text());assert len(out['facilities'])==4;assert len({x['facilityId'] for x in out['facilities']})==4;assert all(not x['manifestApplied'] and not x['adoptionApproved'] for x in out['facilities'])
print('4 facilities / 3 unapproved proposed locations / general raw hashes PASS')
(E/'verification-inputs.json').write_text(json.dumps({'schema':1,'headCommit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'verifiedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'inputSHA256':hashes,'command':'uv run --with shapely --with mapbox-vector-tile==2.2.0 python data/work/kanto-prefecture/gunma/east/verify.py','exitCode':0,'scope':'原本hash、PBF再decode/binding、union再計算、丸め内包、候補schema/4ID coverage。棟同定の人間レビューは代替しない。全suiteは親。'},ensure_ascii=False,indent=2)+'\n')
