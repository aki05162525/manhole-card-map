import json,zipfile,csv,io,sys,hashlib
rows=json.load(open('data/evidence/kanto-prefecture/tokyo/west/retrievals.json'))
def table(ds,c):
    u=f'https://data.address-br.digital.go.jp/{ds}/city/{ds}_city{c}.csv.zip'
    r=[x for x in rows if x['url']==u and x.get('rawPath')][-1]
    z=zipfile.ZipFile(r['rawPath'])
    n=z.namelist()[0]
    raw=z.read(n)
    return list(csv.DictReader(io.StringIO(raw.decode('utf-8-sig')))),r['sha256'],n
targets={'koganei':('132101','本町','1','18','14'),'hino':('132128','日野本町','1','7','2'),'musashino':('132039','境','1','1','7'),'toei':('131202','東大泉','2','10','5'),'ogikubo':('131156','荻窪','2','43','36')}
for k,(c,oaza,ch,blk,rsdt) in targets.items():
    town,th,_=table('mt_town',c)
    tw=[t for t in town if t.get('oaza_cho')==oaza and str(t.get('chome_number') or '').lstrip('0')==ch or (t.get('oaza_cho')==oaza and t.get('chome')in(ch,))]
    print(k,'town',[(t.get('machiaza_id'),t.get('oaza_cho'),t.get('chome'),t.get('chome_number'),t.get('rsdt_addr_flg')) for t in tw][:4], list(town[0].keys())[:30] if not tw else '')
    ids={t['machiaza_id'] for t in tw}
    for t in tw[:1]: pass
    if k=='hino':
        pa,_,_=table('mt_parcel',c)
        print('  parcelkeys',list(pa[0].keys())[:14])
        hit=[x for x in pa if x['machiaza_id'] in ids and x.get('prc_num1','').lstrip('0')==blk]
        print('  parcels',[(x['prc_id'],x['prc_num1'],x['prc_num2'],x['prc_num3']) for x in hit][:80])
        pos,_,_=table('mt_parcel_pos',c)
        idx={(x['machiaza_id'],x['prc_id']):x for x in pos}
        for x in hit:
            q=idx.get((x['machiaza_id'],x['prc_id']))
            if q: print('   ',x['prc_num1'],x['prc_num2'],x['prc_num3'],q['rep_lat'],q['rep_lon'],q['rep_srid'],q.get('rep_scale'),q.get('rep_src_code'),q.get('prc_data_mnt_date'))
        continue
    blks,bh,_=table('mt_rsdtdsp_blk',c)
    b=[x for x in blks if x['machiaza_id'] in ids and x['blk_num'].lstrip('0')==blk]
    print('  blk',[(x['machiaza_id'],x['blk_id'],x['blk_num'],x.get('status_flg'),x.get('efct_date'),x.get('ablt_date')) for x in b])
    bid={(x['machiaza_id'],x['blk_id']) for x in b}
    rs,rh,_=table('mt_rsdtdsp_rsdt',c)
    a=[x for x in rs if (x['machiaza_id'],x['blk_id']) in bid and x['rsdt_num'].lstrip('0')==rsdt]
    print('  rsdt',[(x['rsdt_id'],x['rsdt2_id'],x['rsdt_num'],x.get('rsdt_num2'),x.get('status_flg'),x.get('efct_date'),x.get('ablt_date')) for x in a])
    pos,ph,fn=table('mt_rsdtdsp_rsdt_pos',c)
    key={(x['machiaza_id'],x['blk_id'],x['rsdt_id'],x['rsdt2_id']) for x in a}
    pp=[x for x in pos if (x['machiaza_id'],x['blk_id'],x['rsdt_id'],x['rsdt2_id']) in key]
    print('  pos',fn,ph[:12],pp)
    bp,_,_=table('mt_rsdtdsp_blk_pos',c)
    print('  blkpos',[x for x in bp if (x['machiaza_id'],x['blk_id']) in bid][:1])
    print('  rsdt-in-block', sorted({x['rsdt_num'] for x in rs if (x['machiaza_id'],x['blk_id']) in bid},key=lambda s:int(s) if s.isdigit() else 0)[:60])
