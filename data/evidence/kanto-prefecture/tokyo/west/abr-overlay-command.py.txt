import json,zipfile,csv,io,sys,math
from PIL import Image,ImageDraw,ImageFont
rows=json.load(open('data/evidence/kanto-prefecture/tokyo/west/retrievals.json'))
def table(ds,c):
    u=f'https://data.address-br.digital.go.jp/{ds}/city/{ds}_city{c}.csv.zip'
    r=[x for x in rows if x['url']==u and x.get('rawPath')][-1]
    z=zipfile.ZipFile(r['rawPath']);n=z.namelist()[0]
    return list(csv.DictReader(io.StringIO(z.read(n).decode('utf-8-sig'))))
key,c,mz,blk,lat,lng,half=sys.argv[1],sys.argv[2],sys.argv[3],sys.argv[4],float(sys.argv[5]),float(sys.argv[6]),float(sys.argv[7])
pos=table('mt_rsdtdsp_rsdt_pos',c);rs=table('mt_rsdtdsp_rsdt',c)
num={(x['machiaza_id'],x['blk_id'],x['rsdt_id'],x['rsdt2_id']):x for x in rs}
W='data/work/kanto-prefecture/tokyo/west/'
cb=json.load(open(W+key+'-combined.json'))
n=2**16;e=cb['extent'];bx,by=cb['baseTile']['x'],cb['baseTile']['y']
xy=lambda la,ln:(((ln+180)/360*n-bx)*e,((1-math.asinh(math.tan(math.radians(la)))/math.pi)/2*n-by)*e)
cx,cy=xy(lat,lng);s=1200/(2*half);tr=lambda p:((p[0]-cx+half)*s,(p[1]-cy+half)*s)
im=Image.new('RGB',(1200,1200),'white');d=ImageDraw.Draw(im);f=ImageFont.load_default(size=12)
for i,ft in enumerate(cb['features']):
    if ft.get('layer') not in('BldA','RdCL','RailCL'):continue
    g=ft['geometry'];co=g['coordinates']
    rr=[r for p in co for r in p] if g['type']=='MultiPolygon' else co if g['type'] in('Polygon','MultiLineString') else [co]
    for r in rr:
        q=[tr(p) for p in r]
        if ft['layer']=='BldA':
            if len(q)>2:d.polygon(q,fill=(219,231,237),outline=(40,40,40))
        else:d.line(q,fill=(240,140,0),width=2)
    if ft['layer']=='BldA' and g['type']=='Polygon':
        r0=co[0];d.text(tr((sum(p[0] for p in r0)/len(r0),sum(p[1] for p in r0)/len(r0))),str(i),fill=(0,110,0),font=f)
out=[]
for p in pos:
    if p['machiaza_id']==mz and abs(float(p['rep_lat'])-lat)<0.004 and abs(float(p['rep_lon'])-lng)<0.005:
        k=(p['machiaza_id'],p['blk_id'],p['rsdt_id'],p['rsdt2_id']);a=num.get(k)
        if not a: continue
        la,ln=float(p['rep_lat']),float(p['rep_lon']);x,y=tr(xy(la,ln))
        if 0<=x<1200 and 0<=y<1200:
            lab=f"{a['blk_num']}-{a['rsdt_num']}"
            col='red' if p['blk_id'].lstrip('0')==blk else 'blue'
            d.ellipse([x-3,y-3,x+3,y+3],fill=col);d.text((x+4,y-6),lab,fill=col,font=f)
            out.append((lab,la,ln,p['rep_scale'],p['rsdt_addr_mtd_code'],p['rsdt_addr_data_mnt_date']))
im.save(f'data/evidence/kanto-prefecture/tokyo/west/{key}/abr-overlay.png')
print(sorted(out)[:40])
