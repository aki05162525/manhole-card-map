import sys,json,math,io
sys.path.insert(0,'tools/national')
from research_common import fetch_cached
from PIL import Image,ImageDraw
E='data/evidence/kanto-prefecture/tokyo/west/';W='data/work/kanto-prefecture/tokyo/west/'
UA={'User-Agent':'Mozilla/5.0'}
key,lat,lng,zoom=sys.argv[1],float(sys.argv[2]),float(sys.argv[3]),18
span=int(sys.argv[4]) if len(sys.argv)>4 else 1
n=2**zoom
fx=(lng+180)/360*n;fy=(1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*n
tx,ty=int(fx),int(fy)
xs=range(tx-span,tx+span+1);ys=range(ty-span,ty+span+1)
im=Image.new('RGB',(256*len(xs),256*len(ys)),'white')
rows=json.load(open(E+'retrievals.json'))
for iy,y in enumerate(ys):
    for ix,x in enumerate(xs):
        u=f'https://cyberjapandata.gsi.go.jp/xyz/seamlessphoto/{zoom}/{x}/{y}.jpg'
        raw,rec=fetch_cached(u,W+'raw','tokyo-west-2026-11-06',headers=UA,timeout=30)
        rec['facilityKey']=key+'-photo';rows.append(rec)
        if raw: im.paste(Image.open(io.BytesIO(raw)).convert('RGB'),(ix*256,iy*256))
json.dump(rows,open(E+'retrievals.json','w'),ensure_ascii=False,indent=2)
c=json.load(open(W+key+'-combined.json'))
bx,by,e=c['baseTile']['x'],c['baseTile']['y'],c['extent']
to=lambda p:(((bx+p[0]/e)*(2**(zoom-16))-xs[0])*256,((by+p[1]/e)*(2**(zoom-16))-ys[0])*256)
d=ImageDraw.Draw(im)
for f in c['features']:
    if f.get('layer')!='BldA':continue
    g=f['geometry'];rr=[r for p in g['coordinates'] for r in p] if g['type']=='MultiPolygon' else g['coordinates']
    for r in rr:
        q=[to(p) for p in r]
        if len(q)>2 and any(0<=a<im.width and 0<=b<im.height for a,b in q): d.line(q+[q[0]],fill=(255,240,0),width=1)
im.save(W+key+'-photo.png');print(im.size,[ (r['url'].split('/')[-1],r.get('status')) for r in rows[-1:]])
