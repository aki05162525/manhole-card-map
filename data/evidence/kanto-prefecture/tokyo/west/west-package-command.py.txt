def write(p, x):
    p.write_text(json.dumps(x, ensure_ascii=False, indent=2) + '\n')


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def rowsha(x):
    return hashlib.sha256(json.dumps(x, sort_keys=True, ensure_ascii=False).encode()).hexdigest()


def source(url):
    hit = [r for r in retrievals if r['url'] == url]
    if not hit:
        raise SystemExit('not retrieved: ' + url)
    r = hit[-1]
    return {k: r.get(k) for k in ('url', 'status', 'httpStatus', 'fetchedAt', 'attemptedAt', 'sha256', 'etag', 'lastModified', 'rawPath', 'error') if r.get(k) is not None}


def seconds(key):
    return sum(i['seconds'] for i in timing['intervals'] if i['key'] == key)
