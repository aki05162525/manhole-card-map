#!/usr/bin/env python3
"""calc.py <prefix> <facilityId> <featureIndices comma> <lat> <lng>: strict inclusion on selected fragment(s), round to 7dp; writes evidence <prefix>-calc.json"""
import sys,json,math,datetime
sys.path.insert(0,'tools/national')
from research_common import rounded_coordinate,contains_point
from shapely.geometry import shape,Point
from shapely.ops import unary_union
prefix,fid,idx,lat,lng=sys.argv[1],sys.argv[2],[int(x) for x in sys.argv[3].split(',')],float(sys.argv[4]),float(sys.argv[5])
c=json.load(open(f'data/work/kanto-prefecture/tokyo/south/{prefix}-combined.json'));b=c['baseTile'];n=2**16;ex=c['extent']
xy=(((lng+180)/360*n-b['x'])*ex,((1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*n-b['y'])*ex)
scale=40075016.68557849*math.cos(math.radians(lat))/(n*ex)
fs=[c['features'][i] for i in idx]
for f in fs: assert f['layer']=='BldA'
primary=fs[0]
res=rounded_coordinate(primary['geometry'],b,xy)
shapes=[shape(f['geometry']) for f in fs]
valid=[s.is_valid for s in shapes];U=unary_union([s.buffer(0.5) for s in shapes])
P=shape(primary['geometry']);minx,miny,maxx,maxy=P.bounds
others=[]
for i,f in enumerate(c['features']):
    if f['layer']=='BldA' and i not in idx and contains_point(f['geometry'],xy): others.append(i)
out=dict(schema=1,facilityId=fid,checkedAt='2026-11-05',computedAt=datetime.datetime.now(datetime.timezone.utc).isoformat(),source='国土地理院最適化ベクトルタイル PMTiles',sourceURL='https://cyberjapandata.gsi.go.jp/xyz/optimal_bvmap-v1/optimal_bvmap-v1.pmtiles',baseTile=b,selectedFeatureIndices=idx,
 fragments=[dict(combinedIndex=i,sourceTile=f['sourceTile'],sourceFeatureIndex=f['sourceFeatureIndex'],attributes={k:v for k,v in f.items() if k not in('geometry','sourceTile','sourceFeatureIndex','tileBoundaryTouched','outerBoundaryTouched','layer')},tileBoundaryTouched=f['tileBoundaryTouched'],outerBoundaryTouched=f['outerBoundaryTouched'],valid=v,areaApproxM2=round(s.area*scale**2,1),geometry=f['geometry']) for i,f,s,v in zip(idx,fs,shapes,valid)],
 totalAreaApproxM2=round(sum(s.area for s in shapes)*scale**2,1),primaryBoundsApproxM=[round((maxx-minx)*scale,1),round((maxy-miny)*scale,1)],
 manualPointInputLatLng=[lat,lng],manualPointXY=xy,inclusion=res,otherBldAContainingPoint=others,minDistanceToPrimaryEdgeApproxM=round(P.exterior.distance(Point(xy))*scale,1),acquiredBounds=c['bounds'],scale_m_per_tile_unit=scale,processing='decode mapbox-vector-tile 2.2.0 y down; combine_tiles; manual point; Web Mercator inverse without datum transformation; 7 decimal rounding',crsApproved=False,identityApproved=False,adoptionApproval=False)
open(f'data/evidence/kanto-prefecture/tokyo/south/{prefix}-calc.json','w').write(json.dumps(out,ensure_ascii=False,indent=2)+'\n')
print(prefix,res['coordinate'],'in',res['unroundedWithin'],res['roundedWithin'],'others',others,'edge m',out['minDistanceToPrimaryEdgeApproxM'],'areas',[f['areaApproxM2'] for f in out['fragments']],'bbox',out['primaryBoundsApproxM'],'bdy',[(f['tileBoundaryTouched'],f['outerBoundaryTouched'],f['valid']) for f in out['fragments']],[f['sourceFeatureIndex'] for f in out['fragments']],[f['sourceTile'] for f in out['fragments']])
