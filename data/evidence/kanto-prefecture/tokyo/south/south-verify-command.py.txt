log=[];ok=True
def chk(name,cond,detail=''):
    global ok;ok=ok and bool(cond);log.append(f"{'PASS' if cond else 'FAIL'} {name} {detail}")
sha=lambda b:hashlib.sha256(b).hexdigest()
rowsha=lambda x:sha(json.dumps(x,sort_keys=True,ensure_ascii=False).encode())
L=json.load(open(E/'retrievals.json'))
bad=[r['url'] for r in L if r.get('sha256') and sha(Path(r['rawPath']).read_bytes())!=r['sha256']]
chk('raw hash of retrievals',not bad,f"{sum(1 for r in L if r.get('sha256'))} acquired; failed attempts kept {sum(1 for r in L if r.get('status')=='failed')}")
for pf in['isle','cadan','niji','oi','reku','reku2']:
    t=json.load(open(E/f'{pf}-tiles.json'))
    fine=True;n=0
    for x in t['tiles']:
        b=Path(x['path']).read_bytes();fine&=sha(b)==x['sha256'];d=decode_tile(b);n+=len(d['BldA']['features']);fine&=len(d['BldA']['features'])==x['layers']['BldA']
    chk(f'{pf} tile hash+redecode',fine,f"{len(t['tiles'])} tiles BldA={n} failures={t['rangeFailures']} etags={t['rangeEtags']}")
    chk(f'{pf} range metadata',t['rangeFailures']==0 and len(t['rangeEtags'])==1 and all(r.get('status') in('acquired','cached') and r.get('contentRange') for r in json.load(open(E/f'{pf}-range-retrievals.json'))))
loc=json.load(open(N/'locations-proposal.json'))['updates'];life=json.load(open(N/'lifecycle-proposal.json'));sel=json.load(open(N/'selection-proposal.json'))
inv=json.load(open('data/national/kanto-prefecture/tokyo/inventory.json'));ids={inv['rows'][i]['facilityId'] for i in(6,11,13,15,19)}
out=json.load(open(N/'outcomes.json'));chk('outcomes has exactly assigned IDs',{r['facilityId'] for r in out['rows']}==ids)
chk('proposal IDs subset',{u['facilityId'] for u in loc}<=ids,str(len(loc)))
chk('all unapproved',sel['approval'] is False and sel['manifestApplied'] is False and all(r['adoptionApproval'] is False for r in out['rows']))
chk('selection hashes',all(a['sha256']==rowsha(next(u for u in loc if u['facilityId']==a['facilityId'])) for a in sel['proposedManifestAppend']['locations'][0]['approved']) and sel['proposedManifestAppend']['locationLifecycle'][0]['sha256']==sha((N/'lifecycle-proposal.json').read_bytes()))
chk('binding hashes',all(b['adoptionSHA256']==rowsha(b['adoption']) and b['identity']['name']==next(r for r in inv['rows'] if r['facilityId']==b['facilityId'])['baselinePlace']['name'] for b in life['bindings']))
for u in loc:
    p=Path(u['evidence']['snapshotPath']);c=json.loads(p.read_text());chk('snapshot hash '+p.name,sha(p.read_bytes())==u['evidence']['snapshotSHA256'])
    combined=json.load(open(W/f"{p.name.replace('-calc.json','')}-combined.json"));b=combined['baseTile'];f=combined['features'][c['selectedFeatureIndices'][0]];la,ln=u['coordinate']['lat'],u['coordinate']['lng']
    x=((ln+180)/360*65536-b['x'])*4096;y=((1-math.asinh(math.tan(math.radians(la)))/math.pi)/2*65536-b['y'])*4096
    h=[i for i,g in enumerate(combined['features']) if g['layer']=='BldA' and contains_point(g['geometry'],(x,y))]
    chk(f"{p.name} rounded point inside only selected",h==c['selectedFeatureIndices'][:1],str(u['coordinate']))
    chk(f"{p.name} geometry unchanged",f['geometry']==c['fragments'][0]['geometry'])
oi=json.load(open(W/'oi-combined.json'));b=oi['baseTile']
for nm,la,ln in[('association-mymaps',35.607687,139.731499),('council-link',35.6076339,139.7314736)]:
    x=((ln+180)/360*65536-b['x'])*4096;y=((1-math.asinh(math.tan(math.radians(la)))/math.pi)/2*65536-b['y'])*4096
    chk('oi auxiliary pin '+nm,[i for i,g in enumerate(oi['features']) if g['layer']=='BldA' and contains_point(g['geometry'],(x,y))]==[5756])
rows=json.load(open(E/'inspection.json'))['rows']
for o in rows:validate_observation(o)
chk('observation schema',len(rows)==5)
forbid=[p.name for d in(E,N) for p in d.rglob('*') if p.suffix.lower() in('.html','.pdf','.jpg','.jpeg','.kml')]
chk('no general originals in evidence/national',not forbid,str(forbid))
chk('work ignored by git',subprocess.run(['git','check-ignore','-q',str(W/'isle-card.html')]).returncode==0)
