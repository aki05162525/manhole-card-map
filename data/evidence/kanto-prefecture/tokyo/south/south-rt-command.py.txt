#!/usr/bin/env python3
"""south group helper: timer / get"""
import sys,json,time,datetime,pathlib
sys.path.insert(0,'tools/national')
from research_common import fetch_cached
E_=pathlib.Path('data/evidence/kanto-prefecture/tokyo/south');W=pathlib.Path('data/work/kanto-prefecture/tokyo/south')
UA={'User-Agent':'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36'}
now=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
def timer(cmd,fid=None,act=None):
    p=E_/'timing.json';st=W/'active-timer.json'
    d=json.loads(p.read_text()) if p.exists() else {'schema':1,'scope':'south group; wall time measured live from start of each facility work block; shared intervals are not allocated to facilities','intervals':[]}
    if st.exists():
        a=json.loads(st.read_text());a['seconds']=round(time.time()-a.pop('epoch'),3);a['finishedAt']=now();d['intervals'].append(a);st.unlink();p.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n');print('stopped',a['facilityId'],a['seconds'])
    if cmd=='start':
        st.write_text(json.dumps({'facilityId':fid,'activity':act,'startedAt':now(),'epoch':time.time()},ensure_ascii=False));print('started',fid)
def get(url,key,ua=True):
    raw,r=fetch_cached(url,W/'raw','tokyo-south-2026-11-05',headers=UA if ua else None,timeout=45)
    r=dict(r);r['key']=key
    p=E_/'retrievals.json';L=json.loads(p.read_text()) if p.exists() else []
    if not any(x.get('url')==url and x.get('sha256')==r.get('sha256') and x.get('status')==r.get('status') and x.get('error')==r.get('error') for x in L):
        L.append(r);p.write_text(json.dumps(L,ensure_ascii=False,indent=2)+'\n')
    if raw is not None:
        (W/key).write_bytes(raw)
    print(key,r.get('status'),r.get('httpStatus'),r.get('bytes') or r.get('responseBytes'),r.get('sha256','')[:12],r.get('error',''))
    return raw
if __name__=='__main__':
    if sys.argv[1]=='timer': timer(*sys.argv[2:])
    elif sys.argv[1]=='get':
        for a in sys.argv[2:]:
            k,u=a.split('=',1);get(u,k)
