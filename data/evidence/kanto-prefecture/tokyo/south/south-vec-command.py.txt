#!/usr/bin/env python3
"""usage: vec.py <prefix> <lat> <lng> [radius] : fetch z16 tiles via PMTiles Range, combine, save SVG + per-area PNG"""
import sys,json,math,hashlib,os
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import fetch_cached,adjacent_tiles,decode_tile,combine_tiles,render_svg
from pmtiles.reader import Reader
E=Path('data/evidence/kanto-prefecture/tokyo/south');W=Path('data/work/kanto-prefecture/tokyo/south');url='https://cyberjapandata.gsi.go.jp/xyz/optimal_bvmap-v1/optimal_bvmap-v1.pmtiles'
prefix,lat,lng=sys.argv[1],float(sys.argv[2]),float(sys.argv[3]);radius=int(sys.argv[4]) if len(sys.argv)>4 else 1
records=[];failed=[]
def get_bytes(offset,length):
    b,r=fetch_cached(url,W/'ranges','tokyo-south-2026-11-05',headers={'Range':f'bytes={offset}-{offset+length-1}'},timeout=60)
    records.append(r)
    if b is None:
        failed.append(r);raise ValueError(r)
    return b
reader=Reader(get_bytes)
center=dict(z=16,x=int((lng+180)/360*65536),y=int((1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*65536));tiles=[];summary=[]
for t in adjacent_tiles(center,radius):
    raw=reader.get(t['z'],t['x'],t['y']);
    if raw is None:
        summary.append(dict(tile=t,missing=True));continue
    p=E/f"{prefix}-{t['z']}-{t['x']}-{t['y']}.pbf.gz";p.write_bytes(raw);d=decode_tile(raw);fs=[]
    for name in ['BldA','RdCL','WL','WA','WStrA','RailCL','Anno']:
        if name in d:
            fs.extend(dict(f,layer=name) for f in d[name]['features'] if f['geometry']['type'] in('Polygon','MultiPolygon','LineString','MultiLineString'))
    tiles.append(dict(tile=t,extent=d['BldA']['extent'],features=fs));summary.append(dict(tile=t,path=str(p),sha256=hashlib.sha256(raw).hexdigest(),bytes=len(raw),layers={k:len(v['features']) for k,v in d.items()}))
c=combine_tiles(tiles);(W/f'{prefix}-combined.json').write_text(json.dumps(c));(E/f'{prefix}-full.svg').write_text(render_svg(c))
(E/f'{prefix}-range-retrievals.json').write_text(json.dumps(records,indent=2)+'\n')
etags={r.get('etag') for r in records};st={r.get('status') for r in records}
(E/f'{prefix}-tiles.json').write_text(json.dumps(dict(sourceURL=url,searchSeedOnly=[lat,lng],seedBasis=sys.argv[5] if len(sys.argv)>5 else 'search window only; never adoption evidence',rangeEtags=sorted(x for x in etags if x),rangeStatuses=sorted(st),rangeFailures=len(failed),tiles=summary),indent=2,ensure_ascii=False)+'\n')
print('tiles',len(tiles),'features',len(c['features']),'etags',etags,'status',st,'failed',len(failed),'base',c['baseTile'])
print({k:v for s in summary[:1] for k,v in s.get('layers',{}).items()})
