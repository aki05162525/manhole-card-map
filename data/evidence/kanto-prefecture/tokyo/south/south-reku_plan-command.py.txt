def bil(q,u,v):
    return tuple((1-u)*(1-v)*q['NW'][i]+u*(1-v)*q['NE'][i]+(1-u)*v*q['SW'][i]+u*v*q['SE'][i] for i in(0,1))
def inv(q,p):
    u,v=.5,.5
    for _ in range(50):
        x,y=bil(q,u,v);e=(p[0]-x,p[1]-y)
        du=tuple((bil(q,u+1e-4,v)[i]-bil(q,u-1e-4,v)[i])/2e-4 for i in(0,1));dv=tuple((bil(q,u,v+1e-4)[i]-bil(q,u,v-1e-4)[i])/2e-4 for i in(0,1))
        D=du[0]*dv[1]-du[1]*dv[0];u+=(e[0]*dv[1]-e[1]*dv[0])/D;v+=(du[0]*e[1]-du[1]*e[0])/D
    return u,v
