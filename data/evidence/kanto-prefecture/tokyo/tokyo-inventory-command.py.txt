import json,hashlib,time,datetime,sys
from pathlib import Path
sys.path.insert(0,'tools/national')
from assemble import assemble,read_json
root=Path('.').resolve(); dest=Path('data/national/kanto-prefecture/tokyo'); ev=Path('data/evidence/kanto-prefecture/tokyo')
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def write(p,v):p.write_text(json.dumps(v,ensure_ascii=False,indent=2)+'\n')
ix=read_json(Path('data/national/issue57/remaining-index.json')); ledger,audit=assemble(root,read_json(Path('tools/national/manifest.json')))
selected=[r for r in ix['facilities'] if '東京都' in r['address'] or r['name'] in ['小金井市観光まちおこし協会','コネクトすみだ[まち処]']]
assert len(selected)==24
ps={p['id']:p for p in ledger['places']}; rows=[]; timing=[]; hashes={}; raw=[]
for r in selected:
 start=time.monotonic(); at=datetime.datetime.now(datetime.timezone.utc).isoformat(); id=r['facilityId']; h=[]
 for ref in r['history']:
  p,ptr=ref['ref'].split('#',1); obj=read_json(Path(p)); hashes[p]=sha(p)
  for k in ptr.strip('/').split('/') if ptr.strip('/') else []:obj=obj[int(k)] if isinstance(obj,list) else obj[k]
  h.append(dict(ref=ref['ref'],kind=ref['kind'],value=obj))
 orig=h[0]['value']; attempts=orig.get('attempts',[])
 for a in attempts:
  p=a.get('snapshotPath'); digest=a.get('snapshotSHA256')
  if p:raw.append(dict(facilityId=id,**a,rawAvailable=Path(p).exists(),hashMatches=Path(p).exists() and sha(p)==digest))
 assert 'coordinate' not in ps[id]
 rows.append(dict(facilityId=id,name=r['name'],address=r['address'],assignment='original-ledger-address; omitted prefecture requires operator confirmation' if '東京都' not in r['address'] else 'original-ledger-address',baselinePlace=ps[id],baselineRelations=[rel for rel in ledger['relations'] if rel['placeId']==id],inheritedIndex=r,resolvedHistory=h,stage='history-resolved; current-building-and-footprint-comparison-not-started',adoptionApproval=False))
 timing.append(dict(facilityId=id,activity='resolve indexed JSON pointers and verify historical raw hashes (machine time; excludes earlier shared reading)',startedAt=at,seconds=round(time.monotonic()-start,6)))
for p in ['tools/national/manifest.json','data/national/issue57/remaining-index.json','data/national/issue57/human-confirmation-queue.json']:
 hashes[p]=sha(p)
write(dest/'inventory.json',dict(schema=1,asOf='2026-11-05',baselineCommit='4f5253b8973cb388deecff0bc051f36770e77579',nationalUnlocated=sum('coordinate' not in p for p in ledger['places']),targetCount=len(rows),assignmentStatus='24 provisional; missing-prefecture operator evidence pending; nationwide omission scan pending',inputSHA256=hashes,assemblyInputSHA256=audit['inputSHA256'],rows=rows))
write(ev/'inherited-retrievals.json',raw);write(ev/'timing.json',dict(schema=1,priorSharedReading='not measured; not allocated to facilities',intervals=timing))
print('inventory',len(rows),'raw',len(raw),'missing/hash-mismatch',sum(not r['hashMatches'] for r in raw))
