import sys,json
sys.path.insert(0,sys.argv[1])
from northlib import *
from research_common import contains_point
c=load('akabanedai');fs=c['features'];C=json.load(open(sys.argv[2]))
P={'ラボ41':C[0],'ミュージアム棟':C[1],'スターハウス44':C[2],'スターハウス43':C[3],'スターハウス42':C[4]}
fid={'ラボ41':3660,'スターハウス44':3662,'スターハウス43':3664,'スターハウス42':3692}
s=scale(35.777);print('m/unit',s)
for k,i in fid.items():
 r=outer(fs[i]);print(k,i,'area',round(area(r)*s*s,1),'planPx',P[k]['n'],'tile',fs[i]['sourceTile'],fs[i]['sourceFeatureIndex'],fs[i]['tileBoundaryTouched'])
fitk=['ラボ41','スターハウス44','スターハウス42']
src=lambda k:(P[k]['cx'],P[k]['cy']);dst=lambda k:centroid(outer(fs[fid[k]]))
a,b=fit([src(k) for k in fitk],[dst(k) for k in fitk]);print('units/px',abs(a),'m/px',abs(a)*s,'px/m',1/(abs(a)*s),'rot',rotationDeg(a))
rows=[]
for k in fid:
 p=apply(a,b,src(k));m=dst(k);rows.append(dict(name=k,planCentroidPx=src(k),feature=fid[k],predicted=p,residualUnits=round(math.dist(p,m),1),residualM=round(math.dist(p,m)*s,2),role='fit' if k in fitk else 'holdout'));print(k,rows[-1]['residualM'],'m')
# all four fit for final; leave-one-out spread
a4,b4=fit([src(k) for k in fid],[dst(k) for k in fid]);print('fit4 m/px',abs(a4)*s,'rot',rotationDeg(a4))
mus=P['ミュージアム棟'];mc=apply(a4,b4,(mus['cx'],mus['cy']));print('museum',mc)
loo=[]
for k in fid:
 ks=[x for x in fid if x!=k];a1,b1=fit([src(x) for x in ks],[dst(x) for x in ks]);q=apply(a1,b1,(mus['cx'],mus['cy']));loo.append(dict(left=k,point=q,deltaM=round(math.dist(q,mc)*s,2)));print('LOO',k,loo[-1]['deltaM'])
x0,y0,x1,y1=mus['bbox'];rect=[apply(a4,b4,p) for p in ((x0,y0),(x1,y0),(x1,y1),(x0,y1),(x0,y0))];g=dict(type='Polygon',coordinates=[rect]);print('rect',rect,'area m2',area(rect)*s*s)
hits=[i for i,f in enumerate(fs) if f['layer']=='BldA' and (contains_point(g,centroid(outer(f))) or any(contains_point(g,p) for p in outer(f)))];print('BldA inside',hits)
rd=[i for i,f in enumerate(fs) if f['layer']=='RdCL' and any(contains_point(g,q) for q in (f['geometry']['coordinates'] if f['geometry']['type']=='LineString' else [t for l in f['geometry']['coordinates'] for t in l]))];print('roads inside',rd)
em=[6210.178544074297,4970.3963958323];print('embed center dist m',math.dist(em,mc)*s)
near=nearest(c,mc,4);print(near,[fs[i]['layer'] for d,i in near])
res=rounded_coordinate(g,c['baseTile'],[round(mc[0]),round(mc[1])]);print(res)
json.dump(dict(rows=rows,fit4=dict(metersPerPx=abs(a4)*s,rotationDeg=rotationDeg(a4)),fit3=dict(metersPerPx=abs(a)*s,rotationDeg=rotationDeg(a)),museumCombined=mc,leaveOneOut=loo,museumRectCombined=rect,museumRectAreaM2=area(rect)*s*s,bldAInsideRect=hits,roadsInsideRect=rd,embedCenterCombined=em,embedCenterDistanceM=math.dist(em,mc)*s,point=res,planCompsPx=C),open(sys.argv[3],'w'),ensure_ascii=False,indent=1)
