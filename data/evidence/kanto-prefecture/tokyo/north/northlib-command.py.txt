# Shared numeric helpers for north comparisons. Fit results support, but never replace, the human map reading.
import json,math,sys,cmath
from html import escape
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import rounded_coordinate
E=Path('data/evidence/kanto-prefecture/tokyo/north')
def load(a): return json.load(open(f'data/work/kanto-prefecture/tokyo/north/vector/{a}-combined.json'))
def area(r): return abs(sum(p[0]*q[1]-q[0]*p[1] for p,q in zip(r,r[1:])))/2
def centroid(r):
 A=sum(p[0]*q[1]-q[0]*p[1] for p,q in zip(r,r[1:]))/2
 return [sum((p[0]+q[0])*(p[0]*q[1]-q[0]*p[1]) for p,q in zip(r,r[1:]))/(6*A),sum((p[1]+q[1])*(p[0]*q[1]-q[0]*p[1]) for p,q in zip(r,r[1:]))/(6*A)]
def outer(f):
 g=f['geometry'];return g['coordinates'][0] if g['type']=='Polygon' else max((po[0] for po in g['coordinates']),key=area)
def scale(lat,z=16,ext=4096): return 40075016.68557849*math.cos(math.radians(lat))/(2**z*ext)
def fit(src,dst):
 # complex similarity z'=a z+b ; image y down and tile y down keep the same handedness
 s=[complex(*p) for p in src];d=[complex(*p) for p in dst];sm=sum(s)/len(s);dm=sum(d)/len(d)
 a=sum((x-dm)*(y-sm).conjugate() for x,y in zip(d,s))/sum(abs(y-sm)**2 for y in s);b=dm-a*sm
 return a,b
def apply(a,b,p): z=a*complex(*p)+b;return [z.real,z.imag]
def rotationDeg(a): return math.degrees(cmath.phase(a))
def nearest(c,pt,n=4):
 rows=[]
 for i,f in enumerate(c['features']):
  if f.get('layer')!='BldA':continue
  m=centroid(outer(f));rows.append((math.dist(m,pt),i))
 return sorted(rows)[:n]
def describe(c,i,point):
 f=c['features'][i];r=outer(f);res=rounded_coordinate(f['geometry'],c['baseTile'],point)
 lat=res['coordinate']['lat'];s=scale(lat)
 return dict(featureIndexCombined=i,layer=f['layer'],sourceTile=f['sourceTile'],sourceFeatureIndex=f['sourceFeatureIndex'],featureId=f.get('id'),properties=f.get('properties'),tileBoundaryTouched=f['tileBoundaryTouched'],outerBoundaryTouched=f['outerBoundaryTouched'],vertexCount=len(r),ring=r,areaApproxM2=round(area(r)*s*s,1),centroidCombined=centroid(r),manualPointXY=point,inclusion=res,scaleMetersPerUnit=s)
def svg(c,win,highlights,path,title):
 x0,y0,x1,y1=win;P=[f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="{x0} {y0} {x1-x0} {y1-y0}"><title>{escape(title)}; N up; crop only, not acquisition range</title>']
 for i,f in enumerate(c['features']):
  g=f['geometry'];cs=g['coordinates'];t=g['type'];l=f.get('layer')
  if l not in('BldA','RdCL','WL','WA','RailCL','WStrL'):continue
  rs=[r for po in ([cs] if t=='Polygon' else cs) for r in po] if t in('Polygon','MultiPolygon') else [cs] if t=='LineString' else cs
  if not any(x0<=p[0]<=x1 and y0<=p[1]<=y1 for r in rs for p in r):continue
  d=' '.join('M '+' L '.join(f'{p[0]:g},{p[1]:g}' for p in r)+(' Z' if t in('Polygon','MultiPolygon') else '') for r in rs)
  closed=t in('Polygon','MultiPolygon');fill=('#f7b267' if i in highlights else '#dbe7ed') if closed else 'none'
  st={'RdCL':'#d33','WL':'#06c','WA':'#06c','RailCL':'#080','WStrL':'#06c'}.get(l,'#333')
  P.append(f'<path d="{d}" fill="{fill}" fill-rule="evenodd" stroke="{st}" stroke-width="1" vector-effect="non-scaling-stroke"><title>{i}</title></path>')
 for i,pt in highlights.items():
  if pt:P.append(f'<circle cx="{pt[0]}" cy="{pt[1]}" r="5" fill="#c00"/>')
 P.append('</svg>');Path(path).write_text('\n'.join(P)+'\n')
