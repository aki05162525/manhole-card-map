#!/usr/bin/env python3
# render.py <area> <x0> <y0> <x1> <y1> <out.png> [labels]  — coordinates are combined-frame units; N up
import sys,json
from PIL import Image,ImageDraw
a,x0,y0,x1,y1,out=sys.argv[1],*map(float,sys.argv[2:6]),sys.argv[6];lab=len(sys.argv)>7
c=json.load(open(f'data/work/kanto-prefecture/tokyo/north/vector/{a}-combined.json'))
W=1400;s=W/(x1-x0);H=int((y1-y0)*s);im=Image.new('RGB',(W,H),'white');d=ImageDraw.Draw(im)
P=lambda p:((p[0]-x0)*s,(p[1]-y0)*s)
col=dict(BldA=('#cfe0ea','#222'),RdCL=(None,'#d33'),WL=(None,'#06c'),WA=('#bfe3ff','#06c'),RailCL=(None,'#080'),RailTrCL=(None,'#0a0'),Cntr=(None,'#ddc'),WStrL=(None,'#06c'))
order=['Cntr','WA','WL','RailCL','RailTrCL','RdCL','BldA']
for ln in order:
 for i,f in enumerate(c['features']):
  if f.get('layer')!=ln:continue
  g=f['geometry'];t=g['type'];cs=g['coordinates']
  polys=[cs] if t=='Polygon' else cs if t=='MultiPolygon' else None
  fill,st=col.get(ln,(None,'#999'))
  if polys:
   for po in polys:
    if not any(x0<=p[0]<=x1 and y0<=p[1]<=y1 for p in po[0]):continue
    d.polygon([P(p) for p in po[0]],fill=fill,outline=st)
    for h in po[1:]:d.polygon([P(p) for p in h],fill='white',outline=st)
    if lab and ln=='BldA':
     xs=[p[0] for p in po[0]];ys=[p[1] for p in po[0]]
     d.text(P((sum(xs)/len(xs),sum(ys)/len(ys))),str(i),fill='#a00')
  else:
   ls=[cs] if t=='LineString' else cs
   for l in ls:d.line([P(p) for p in l],fill=st,width=1 if ln!='RdCL' else 2)
im.save(out);print(out,im.size,'units/px',1/s)
