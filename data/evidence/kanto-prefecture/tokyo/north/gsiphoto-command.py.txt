import sys,json,io
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import fetch_cached
from PIL import Image
W=Path('data/work/kanto-prefecture/tokyo/north/raw');E=Path('data/evidence/kanto-prefecture/tokyo/north/retrievals.json')
scope,z,x0,y0,x1,y1,kind,out=sys.argv[1],*map(int,sys.argv[2:7]),sys.argv[7],sys.argv[8]
rec=json.loads(E.read_text());c=None
for x in range(x0,x1+1):
 for y in range(y0,y1+1):
  ext='jpg' if kind=='seamlessphoto' else 'png'
  b,r=fetch_cached(f'https://cyberjapandata.gsi.go.jp/xyz/{kind}/{z}/{x}/{y}.{ext}',W,'tokyo-north-2026-11-05',headers={'User-Agent':'Mozilla/5.0 (manhole-card-map research)'});rec.append(dict(r,scope=scope));
  if b is None:print('fail',x,y,r.get('error'));continue
  if c is None:c=Image.new('RGB',((x1-x0+1)*256,(y1-y0+1)*256),'white')
  c.paste(Image.open(io.BytesIO(b)).convert('RGB'),((x-x0)*256,(y-y0)*256))
E.write_text(json.dumps(rec,ensure_ascii=False,indent=1)+'\n');c.save(out);print(out,c.size)
