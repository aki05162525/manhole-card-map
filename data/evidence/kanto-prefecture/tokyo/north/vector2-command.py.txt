#!/usr/bin/env python3
# Additional z16 window; appends range records and tile summary without overwriting earlier evidence. args: name z x y
import sys,json,hashlib
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import fetch_cached,adjacent_tiles,decode_tile,combine_tiles,render_svg
from pmtiles.reader import Reader
E=Path('data/evidence/kanto-prefecture/tokyo/north');W=Path('data/work/kanto-prefecture/tokyo/north');url='https://cyberjapandata.gsi.go.jp/xyz/optimal_bvmap-v1/optimal_bvmap-v1.pmtiles'
rp=E/'range-retrievals.json';records=json.loads(rp.read_text())
def get_bytes(offset,length):
 b,r=fetch_cached(url,W/'ranges','tokyo-north-2026-11-05',headers={'Range':f'bytes={offset}-{offset+length-1}'})
 records.append(r);rp.write_text(json.dumps(records,ensure_ascii=False,indent=1)+'\n')
 if b is None:raise ValueError(r)
 return b
reader=Reader(get_bytes);name,z,x,y=sys.argv[1],*map(int,sys.argv[2:5]);center=dict(z=z,x=x,y=y);tiles=[];summary=[]
for t in adjacent_tiles(center):
 raw=reader.get(t['z'],t['x'],t['y']);p=E/'tiles'/f"{t['z']}-{t['x']}-{t['y']}.pbf.gz";p.write_bytes(raw);d=decode_tile(raw);fs=[]
 for ln in ['BldA','RdCL','WL','WA','Cntr','RailCL','WStrA','WStrL']:
  if ln in d: fs.extend(dict(f,layer=ln) for f in d[ln]['features'] if f['geometry']['type'] in('Polygon','MultiPolygon','LineString','MultiLineString'))
 tiles.append(dict(tile=t,extent=d['BldA']['extent'] if 'BldA' in d else 4096,features=fs));summary.append(dict(tile=t,path=str(p),sha256=hashlib.sha256(raw).hexdigest(),bytes=len(raw),layers={k:len(v['features']) for k,v in d.items()}))
c=combine_tiles(tiles);(W/'vector'/f'{name}-combined.json').write_text(json.dumps(c));(E/f'{name}-full.svg').write_text(render_svg(c))
tp=E/'tiles.json';T=json.loads(tp.read_text());T['areas'].append(dict(area=name,center=center,seedBasis=sys.argv[5] if len(sys.argv)>5 else 'tile window',featureCount=len(c['features']),tiles=summary));tp.write_text(json.dumps(T,ensure_ascii=False,indent=1)+'\n')
print(name,len(c['features']),{r.get('etag') for r in records},{r['status'] for r in records})
