 if not ok:fail.append(msg)

for r in json.load(open(E/'retrievals.json')):
 if r.get('rawPath') and r.get('sha256'):chk(sha(r['rawPath'])==r['sha256'],'raw '+r['url'][:70])

 px=((lng+180)/360*65536-x)*4096;py=((1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*65536-y)*4096

 chk((fp['sourceTile']['x'],fp['sourceTile']['y'],fp['sourceFeatureIndex'])==(x,y,i),f'{slug} provenance');chk(not fp['tileBoundaryTouched'] and not fp['outerBoundaryTouched'],f'{slug} no tile/outer boundary touch')
