# usage: python outer-photo.py.txt <key> <lat> <lng> <halfWidthMeters> <outPng> [zoom] [labelMinArea] [noOverlay]; GSI seamlessphoto tiles (raw under data/work) with vector footprint overlay for reading only. Photos are reading aids with unknown per-tile date, not adopted position source.
import sys,json,math,io
from pathlib import Path
from PIL import Image,ImageDraw,ImageFont
sys.path.insert(0,'tools/national')
from research_common import fetch_cached
key,lat,lng,hw,out=sys.argv[1],float(sys.argv[2]),float(sys.argv[3]),float(sys.argv[4]),sys.argv[5]
z=int(sys.argv[6]) if len(sys.argv)>6 else 18;minA=float(sys.argv[7]) if len(sys.argv)>7 else 15;over=len(sys.argv)<=8
W=Path('data/work/kanto-prefecture/tokyo/outer');L=Path('data/evidence/kanto-prefecture/tokyo/outer/photo-retrievals.jsonl')
n=2**z;m=40075016.68557849*math.cos(math.radians(lat))/(n*256)
px=(lng+180)/360*n*256;py=(1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*n*256;h=hw/m
x0,x1,y0,y1=int((px-h)//256),int((px+h)//256),int((py-h)//256),int((py+h)//256)
im=Image.new('RGB',((x1-x0+1)*256,(y1-y0+1)*256),'#888')
for tx in range(x0,x1+1):
  for ty in range(y0,y1+1):
    u=f'https://cyberjapandata.gsi.go.jp/xyz/seamlessphoto/{z}/{tx}/{ty}.jpg'
    b,r=fetch_cached(u,W/'photo','tokyo-outer-2026-11-05')
    with L.open('a') as f: f.write(json.dumps({k:r.get(k) for k in('url','status','httpStatus','fetchedAt','attemptedAt','reusedAt','sha256','etag','lastModified','error')}|{'facilityKey':key})+'\n')
    if b: im.paste(Image.open(io.BytesIO(b)).convert('RGB'),((tx-x0)*256,(ty-y0)*256))
S=4 if z<=18 else 2
crop=im.crop((int(px-h-x0*256),int(py-h-y0*256),int(px+h-x0*256),int(py+h-y0*256)));side=crop.size[0];R=1300/side
crop=crop.resize((1300,1300),Image.LANCZOS);d=ImageDraw.Draw(crop)
font=ImageFont.truetype('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf',14)
if over:
    x=json.load(open(W/f'{key}-combined.json'));b=x['baseTile'];ext=x['extent'];bn=2**b['z']
    tr=lambda p:(((b['x']+p[0]/ext)/bn*n*256-(px-h))*R,((b['y']+p[1]/ext)/bn*n*256-(py-h))*R)
    sc=40075016.68557849*math.cos(math.radians(lat))/(bn*ext)
    for i,f in enumerate(x['features']):
        if f['layer']!='BldA':continue
        g=f['geometry'];ps=[g['coordinates']] if g['type']=='Polygon' else g['coordinates']
        for poly in ps:
            pts=[tr(p) for p in poly[0]]
            if all(a[0]<0 or a[0]>1300 or a[1]<0 or a[1]>1300 for a in pts):continue
            d.line(pts,fill='#ffee00',width=1)
            A=abs(sum(a[0]*c[1]-c[0]*a[1] for a,c in zip(poly[0],poly[0][1:])))/2*sc*sc
            if A>=minA:
                mx=sum(p[0] for p in pts[:-1])/(len(pts)-1);my=sum(p[1] for p in pts[:-1])/(len(pts)-1);d.text((mx-10,my-7),str(i),fill='#ff3030',font=font)
L5=50/m*R;d.line((20,1280,20+L5,1280),fill='white',width=4);d.text((20,1258),'50 m N-up',fill='white',font=font)
crop.save(out);print('ok',x1-x0+1,y1-y0+1,'tiles z',z)
