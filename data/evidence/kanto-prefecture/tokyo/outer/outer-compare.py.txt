# usage: python outer-compare.py.txt <key>; reads <key>-reading.json (hand-written observations) and computes polygon/rounding checks into <key>-comparison.json. Arithmetic only: identity, CRS and adoption stay unapproved.
import sys,json,math,hashlib
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import rounded_coordinate
key=sys.argv[1];E=Path('data/evidence/kanto-prefecture/tokyo/outer');R=json.loads((E/f'{key}-reading.json').read_text())
x=json.load(open(f'data/work/kanto-prefecture/tokyo/outer/{key}-combined.json'));b=x['baseTile'];ext=x['extent'];n=2**b['z']
out=dict(schema=1,facilityId=R['facilityId'],checkedAt='2026-11-05',source='国土地理院最適化ベクトルタイル PMTiles',sourceURL='https://github.com/gsi-cyberjapan/optimal_bvmap',tileRecord=f'{key}-tiles.json',baseTile=b,fullRender=f'{key}-full.svg',processing='manhole-card-map: decode mapbox-vector-tile 2.2.0, y down, combine_tiles; feature selected manually after reading official/secondary sources; Web Mercator inverse without datum transformation; 7 decimal rounding',reading=R['reading'],decision=R['decision'],adoptionApproval=False,manifestApplied=False,rights=dict(pmtiles='出典: 国土地理院最適化ベクトルタイルをmanhole-card-mapが加工。PDL1.0条件は既存Tokyo証跡(../retrievals.json)を参照し再取得していない。試験公開で基本測量成果ではない。',generalOriginals='自治体・第三者HTML/PDF/画像全文はdata/workのみ。'),crs=dict(tileGrid='extent 4096 XYZ arithmetic',productSpecificDatum='PMTiles metadata(tiles.json)に測地系宣言なし。一般タイル仕様JGD2011の製品適用は未承認。',approved=False))
features=[]
for c in R.get('candidateFeatures',[]):
    f=x['features'][c['combinedIndex']];assert f['layer']=='BldA'
    if (f['sourceTile'],f['sourceFeatureIndex'])!=(c['sourceTile'],c['sourceFeatureIndex']): raise SystemExit('feature identity mismatch')
    ring=f['geometry']['coordinates'][0] if f['geometry']['type']=='Polygon' else f['geometry']['coordinates'][0][0]
    area=abs(sum(a[0]*d[1]-d[0]*a[1] for a,d in zip(ring,ring[1:])))/2
    y=sum(p[1] for p in ring)/len(ring);lat=math.degrees(math.atan(math.sinh(math.pi*(1-2*(b['y']+y/ext)/n))))
    s=40075016.68557849*math.cos(math.radians(lat))/(n*ext)
    features.append(dict(role=c['role'],sourceFeatureIdentity=f"BldA array index {f['sourceFeatureIndex']} in z16/{f['sourceTile']['x']}/{f['sourceTile']['y']}; combined index {c['combinedIndex']}; id={f.get('id')} is not unique",properties=f.get('properties'),geometry=f['geometry'],areaApproxM2=round(area*s*s,1),tileBoundaryTouched=f['tileBoundaryTouched'],outerBoundaryTouched=f['outerBoundaryTouched']))
    if 'pointXY' in c:
        res=rounded_coordinate(f['geometry'],b,c['pointXY'],extent=ext);features[-1].update(manualPointXY=c['pointXY'],inclusion=res);print(key,c['role'],res['coordinate'],res['mechanicalChecksPassed'],round(area*s*s,1),f['tileBoundaryTouched'],f['outerBoundaryTouched'])
out['features']=features;out['nonCandidateFeaturesConsidered']=R.get('nonCandidateFeaturesConsidered',[])
(E/f'{key}-comparison.json').write_text(json.dumps(out,ensure_ascii=False,indent=1)+'\n')
