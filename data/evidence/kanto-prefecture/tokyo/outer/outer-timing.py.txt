# usage: python outer-timing.py.txt <group|facility-key> <start|stop> [phase]; appends measured wall-clock events, never back-filled.
import sys,json,datetime
from pathlib import Path
p=Path('data/evidence/kanto-prefecture/tokyo/outer/timing-events.jsonl')
with p.open('a') as f: f.write(json.dumps(dict(key=sys.argv[1],event=sys.argv[2],phase=(sys.argv[3] if len(sys.argv)>3 else ''),at=datetime.datetime.now(datetime.timezone.utc).isoformat()),ensure_ascii=False)+'\n')
