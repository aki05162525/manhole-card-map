# usage: python outer-vector.py.txt <key> <seedLat> <seedLng> [radius]; seed is a search window only, never adoption evidence.
import sys,json,math,hashlib
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import fetch_cached,adjacent_tiles,decode_tile,combine_tiles,render_svg
from pmtiles.reader import Reader
key,lat,lng=sys.argv[1],float(sys.argv[2]),float(sys.argv[3]);radius=int(sys.argv[4]) if len(sys.argv)>4 else 1
E=Path('data/evidence/kanto-prefecture/tokyo/outer');W=Path('data/work/kanto-prefecture/tokyo/outer')
url='https://cyberjapandata.gsi.go.jp/xyz/optimal_bvmap-v1/optimal_bvmap-v1.pmtiles';records=[]
def get_bytes(offset,length):
    b,r=fetch_cached(url,W/'ranges','tokyo-outer-2026-11-05',headers={'Range':f'bytes={offset}-{offset+length-1}'})
    records.append(r)
    (E/f'{key}-range-retrievals.json').write_text(json.dumps(records,indent=2)+'\n')
    if b is None: raise ValueError(r)
    return b
reader=Reader(get_bytes)
header=reader.header();meta=reader.metadata()
center=dict(z=16,x=int((lng+180)/360*65536),y=int((1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*65536));tiles=[];summary=[]
for t in adjacent_tiles(center,radius):
    raw=reader.get(t['z'],t['x'],t['y'])
    if raw is None:
        summary.append(dict(tile=t,missing=True));continue
    p=E/f"{key}-{t['z']}-{t['x']}-{t['y']}.pbf.gz";p.write_bytes(raw);d=decode_tile(raw);fs=[]
    ext=None
    for name in ['BldA','RdCL','RailCL','WA','WL','Cntr','SpcfArea','Anno','AdmBdry']:
        if name in d:
            ext=d[name]['extent']
            if name in('BldA','RdCL','RailCL','WL','WA'):
                fs.extend(dict(f,layer=name) for f in d[name]['features'] if f['geometry']['type'] in('Polygon','MultiPolygon','LineString','MultiLineString'))
    tiles.append(dict(tile=t,extent=ext or 4096,features=fs));summary.append(dict(tile=t,path=str(p),sha256=hashlib.sha256(raw).hexdigest(),layers={k:len(v['features']) for k,v in d.items()}))
c=combine_tiles(tiles);(W/f'{key}-combined.json').write_text(json.dumps(c));(E/f'{key}-full.svg').write_text(render_svg(c))
(E/f'{key}-tiles.json').write_text(json.dumps(dict(sourceURL=url,cacheVersion='tokyo-outer-2026-11-05',searchSeedOnly=[lat,lng],seedBasis='search window only; never adoption evidence',pmtilesHeader={k:str(v) for k,v in header.items() if k in('tile_type','tile_compression','min_zoom','max_zoom')},metadataKeys=sorted(meta) if isinstance(meta,dict) else None,metadataAttribution=(meta or {}).get('attribution') if isinstance(meta,dict) else None,metadataVersion=(meta or {}).get('version') if isinstance(meta,dict) else None,metadataDescription=(meta or {}).get('description') if isinstance(meta,dict) else None,tiles=summary),ensure_ascii=False,indent=2)+'\n')
print('tiles',len(tiles),'features',len(c['features']),'base',c['baseTile'],'bounds',c['bounds'])
