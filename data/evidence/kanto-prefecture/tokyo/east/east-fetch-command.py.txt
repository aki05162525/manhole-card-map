#!/usr/bin/env python3
# fetch.py <scope> <url>... : data/work/.../east/raw に保存し east/retrievals.json へ追記
import sys,json
from pathlib import Path
from urllib.parse import urlsplit,quote,urlunsplit
sys.path.insert(0,'tools/national')
from research_common import fetch_cached
W=Path('data/work/kanto-prefecture/tokyo/east/raw');E=Path('data/evidence/kanto-prefecture/tokyo/east/retrievals.json')
rec=json.loads(E.read_text()) if E.exists() else []
scope=sys.argv[1]
H={'User-Agent':'Mozilla/5.0 (manhole-card-map research)'}
for url in sys.argv[2:]:
    s=urlsplit(url);u=urlunsplit((s.scheme,s.netloc,quote(s.path,safe='/%:@'),s.query,''))
    b,r=fetch_cached(u,W,'tokyo-east-2026-11-05-ua',headers=H,timeout=30)
    r=dict(r,scope=scope);rec.append(r)
    print(r['status'],r.get('httpStatus'),r.get('bytes'),r.get('contentType'),r.get('sha256') or r.get('responseSHA256'),u,r.get('error',''))
E.write_text(json.dumps(rec,ensure_ascii=False,indent=1)+'\n')
