import json,re,html,sys
n=int(sys.argv[1]);keys=sys.argv[2:]
recs=json.load(open('data/evidence/kanto-prefecture/tokyo/east/retrievals.json'))
for r in recs[-n:]:
    p=r.get('rawPath')
    if not p: print('FAILED',r['url'],r.get('httpStatus'));continue
    if 'html' not in (r.get('contentType') or ''): print('BIN',r['url'],r.get('contentType'));continue
    s=open(p,'rb').read()
    try:t=s.decode('utf8')
    except:t=s.decode('cp932','replace')
    print('=====',r['url'],r.get('lastModified'));print('IMG',[x for x in re.findall(r'<(?:img|iframe)[^>]+src="([^"]+)"',t) if re.search(r'map|access|floor|guide|zu|google|plan|annai|shisetsu',x,re.I)][:25]);print('HREF',[h for h in re.findall(r'href="([^"]+)"',t) if re.search(r'map|google|goo\.gl|floor|\.pdf|access',h,re.I)][:20])
    b=re.sub(r'<script.*?</script>|<style.*?</style>','',t,flags=re.S);b=re.sub(r'<[^>]+>',' ',b);b=re.sub(r'\s+',' ',html.unescape(b))
    for k in keys:
        for m in list(re.finditer(k,b))[:3]: print(' ',k,':',b[max(0,m.start()-110):m.end()+160])
