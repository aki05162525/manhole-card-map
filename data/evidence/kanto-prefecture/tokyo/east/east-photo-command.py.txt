#!/usr/bin/env python3
# photo.py <key> <cx> <cy> <half> <out.png> [marks x,y...] : GSI seamlessphoto z18 + BldA/RdCL overlay
import sys,json,math,io
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import fetch_cached
from PIL import Image,ImageDraw,ImageFont
key,cx,cy,half,out=sys.argv[1],float(sys.argv[2]),float(sys.argv[3]),float(sys.argv[4]),sys.argv[5]
marks=[tuple(map(float,m.split(','))) for m in sys.argv[6:]]
W=Path('data/work/kanto-prefecture/tokyo/east');E=Path('data/evidence/kanto-prefecture/tokyo/east/photo-retrievals.json')
x=json.load(open(W/f'{key}-combined.json'));b=x['baseTile']
# global z16 tile-unit -> z19 pixel
Z=18;f=2**(Z-16)*256/4096
gx0=(b['x']*4096+cx-half)*f;gy0=(b['y']*4096+cy-half)*f;gx1=(b['x']*4096+cx+half)*f;gy1=(b['y']*4096+cy+half)*f
tx0,ty0,tx1,ty1=int(gx0//256),int(gy0//256),int(gx1//256),int(gy1//256)
mos=Image.new('RGB',((tx1-tx0+1)*256,(ty1-ty0+1)*256),'gray');recs=json.loads(E.read_text()) if E.exists() else []
for ty in range(ty0,ty1+1):
    for tx in range(tx0,tx1+1):
        url=f'https://cyberjapandata.gsi.go.jp/xyz/seamlessphoto/{Z}/{tx}/{ty}.jpg'
        raw,r=fetch_cached(url,W/'photo','seamlessphoto-2026-11-05');recs.append(dict(r,scope=key))
        if raw: mos.paste(Image.open(io.BytesIO(raw)).convert('RGB'),((tx-tx0)*256,(ty-ty0)*256))
        else: print('FAILED',url,r.get('error'))
E.write_text(json.dumps(recs,indent=1)+'\n')
ox,oy=tx0*256,ty0*256;box=(int(gx0-ox),int(gy0-oy),int(gx1-ox),int(gy1-oy));S=1200;sc=S/(box[2]-box[0])
im=mos.crop(box).resize((S,S),Image.LANCZOS);d=ImageDraw.Draw(im);font=ImageFont.truetype('/usr/share/fonts/opentype/ipafont-gothic/ipag.ttf',14)
P=lambda p:(((p[0]+b['x']*4096)*f-gx0)*sc,((p[1]+b['y']*4096)*f-gy0)*sc)
for i,ft in enumerate(x['features']):
    g=ft['geometry']
    if ft['layer']=='BldA':
        rs=g['coordinates'] if g['type']=='Polygon' else [r for p in g['coordinates'] for r in p]
        if any(abs(p[0]-cx)<half*1.2 and abs(p[1]-cy)<half*1.2 for p in rs[0]):
            for r in rs: d.line([P(p) for p in r+[r[0]]],fill=(255,255,0),width=1)
            q=P((sum(p[0] for p in rs[0])/len(rs[0]),sum(p[1] for p in rs[0])/len(rs[0])))
            if half<=450: d.text(q,str(i),fill=(255,255,255),font=font,anchor='mm',stroke_width=1,stroke_fill=(0,0,0))
for m in marks:
    q=P(m);d.ellipse([q[0]-6,q[1]-6,q[0]+6,q[1]+6],outline=(255,0,255),width=3)
im.save(out);print(out)
