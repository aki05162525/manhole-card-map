import json,hashlib,glob,sys,gzip,subprocess,datetime
from pathlib import Path
sys.path.insert(0,'tools/national')
from research_common import decode_tile,validate_observation,contains_point,rounded_coordinate
E=Path('data/evidence/kanto-prefecture/tokyo/east');N=Path('data/national/kanto-prefecture/tokyo/east');R={};fails=[]
def chk(n,ok,d=None):
    R[n]=dict(ok=bool(ok),detail=d)
    if not ok: fails.append(n)
    print(('PASS' if ok else 'FAIL'),n,d if d is not None else '')
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
inv=json.load(open('data/national/kanto-prefecture/tokyo/inventory.json'));rows=inv['rows'];ids=[rows[i]['facilityId'] for i in (1,2,16,17,23)]
out=json.load(open(N/'outcomes.json'));chk('assigned-rows-exact',sorted(r['facilityId'] for r in out['rows'])==sorted(ids),len(ids))
asg=[g for g in json.load(open('data/national/kanto-prefecture/tokyo/assignments.json'))['groups'] if g['group']=='east'][0];chk('assignments-match',sorted(asg['facilityIds'])==sorted(ids))
rs=json.load(open(E/'retrievals.json'));bad=[r['url'] for r in rs if r.get('rawPath') and sha(r['rawPath'])!=r['sha256']];chk('retrieval-raw-hash',not bad,dict(records=len(rs),ok=sum(1 for r in rs if r.get('rawPath')),failedAttempts=sum(1 for r in rs if r.get('status')=='failed'),bad=bad))
ph=json.load(open(E/'photo-retrievals.json'));bad=[r['url'] for r in ph if r.get('rawPath') and sha(r['rawPath'])!=r['sha256']];chk('photo-raw-hash',not bad,len(ph))
for k in ['maruyoshi','macaron','connect','mikawashima']:
    t=json.load(open(E/f'{k}-tiles.json'));bad=[x['path'] for x in t['tiles'] if sha(x['path'])!=x['sha256']];chk(f'{k}-tile-hash',not bad and len(t['tiles'])==9)
    rr=json.load(open(E/f'{k}-range-retrievals.json'));chk(f'{k}-range-provenance',all(x['status'] in('acquired','cached') and x.get('contentRange') for x in rr) and len({x.get('etag') for x in rr})==1,dict(records=len(rr),etags=list({x.get('etag') for x in rr})))
for o in json.load(open(E/'inspection.json'))['observations']: validate_observation(o)
chk('observations-valid',True,5)
loc=json.load(open(N/'locations-proposal.json'))['updates'];chk('location-count-excludes-adachi',len(loc)==4 and ids[4] not in [u['facilityId'] for u in loc])
for u in loc:
    chk('snapshot-hash '+u['facilityId'][-12:],sha(u['evidence']['snapshotPath'])==u['evidence']['snapshotSHA256'],u['coordinate'])
lc=json.load(open(N/'lifecycle-proposal.json'));rh=lambda x:hashlib.sha256(json.dumps(x,sort_keys=True,ensure_ascii=False).encode()).hexdigest();chk('binding-hash',all(rh(b['adoption'])==b['adoptionSHA256'] for b in lc['bindings']) and [b['adoption'] for b in lc['bindings']]==loc)
sel=json.load(open(N/'selection-proposal.json'));chk('selection-unapproved',sel['approval'] is False and sel['manifestApplied'] is False and sel['proposedManifestAppend']['locationLifecycle'][0]['sha256']==sha(N/'lifecycle-proposal.json'))
# 保存PBFからの独立再計算
def tile(k,x,y):return decode_tile(open(E/f'{k}-16-{x}-{y}.pbf.gz','rb').read())
for name,k,x,y,i,base,xy,cmp_ in [('maruyoshi','maruyoshi',58226,25793,1551,(58225,25793),(7580,4125),'maruyoshi'),('macaron','macaron',58226,25788,618,(58225,25787),(7184,4581),'macaron'),('mikawashima-a','mikawashima',58215,25793,695,(58214,25792),(4443,8170),'mikawashima'),('mikawashima-b','mikawashima',58215,25794,60,(58214,25792),(4443,8170),'mikawashima')]:
    f=tile(k,x,y)['BldA']['features'][i];dx,dy=(x-base[0])*4096,(y-base[1])*4096
    g=dict(type=f['geometry']['type'],coordinates=[[[p[0]+dx,p[1]+dy] for p in r] for r in f['geometry']['coordinates']])
    rc=rounded_coordinate(g,dict(z=16,x=base[0],y=base[1]),list(xy));c=json.load(open(E/f'{cmp_}-comparison.json'))['candidate'];chk('recompute '+name,rc['mechanicalChecksPassed'] and rc['coordinate']==c,rc['coordinate'])
a=json.load(open(E/'adachi-supplement.json'));chk('adachi-unchanged',a['existingCandidate']==a['reproducedCoordinate'] and a['supplementToSHA256']==sha('data/evidence/kanto-prefecture/tokyo/adachi-comparison.json'),a['existingCandidate'])
