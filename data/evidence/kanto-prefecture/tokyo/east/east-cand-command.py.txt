import json,sys,math
sys.path.insert(0,'tools/national')
from research_common import rounded_coordinate,contains_point
W='data/work/kanto-prefecture/tokyo/east/'
MPU=lambda lat:40075016.68557849*math.cos(math.radians(lat))/(65536*4096)
def segd(p,a,b):
    dx,dy=b[0]-a[0],b[1]-a[1];t=max(0,min(1,((p[0]-a[0])*dx+(p[1]-a[1])*dy)/(dx*dx+dy*dy)))
    return math.hypot(p[0]-a[0]-t*dx,p[1]-a[1]-t*dy)
def margin(ring,p): return min(segd(p,a,b) for a,b in zip(ring,ring[1:]+ring[:1]) if a!=b)
def area(r): return abs(sum(a[0]*b[1]-b[0]*a[1] for a,b in zip(r,r[1:]+r[:1])))/2
out={}
for key,idx,xy in [('maruyoshi',[2982,7479],(7580,4125)),('macaron',[6109],(7184,4581)),('mikawashima',[6677,9777],(4443,8170)),('adachi',[6761],(6705,7060))]:
    x=json.load(open(W+key+'-combined.json'));rows=[]
    for i in idx:
        f=x['features'][i];rc=rounded_coordinate(f['geometry'],x['baseTile'],list(xy));r=f['geometry']['coordinates'][0];s=MPU(rc['coordinate']['lat'])
        rows.append(dict(combinedIndex=i,sourceTile=f['sourceTile'],sourceFeatureIndex=f['sourceFeatureIndex'],vt_code=f['properties'].get('vt_code'),idField=f.get('id'),tileBoundaryTouched=f['tileBoundaryTouched'],outerBoundaryTouched=f['outerBoundaryTouched'],ring=r,approxAreaM2=round(area(r)*s*s,1),marginToEdgeM=round(margin(r,xy)*s,2),check=rc))
    others=[i for i,f in enumerate(x['features']) if f['layer']=='BldA' and i not in idx and contains_point(f['geometry'],list(xy))]
    out[key]=dict(baseTile=x['baseTile'],bounds=x['bounds'],featureCounts={l:sum(1 for f in x['features'] if f['layer']==l) for l in sorted(set(f['layer'] for f in x['features']))},pointTileXY=list(xy),features=rows,alsoContainingOtherBldA=others,metersPerUnit=round(s,4))
x=json.load(open(W+'connect-combined.json'));b=x['baseTile'];s=MPU(35.71)
f=x['features'][5948];l=f['geometry']['coordinates'];A=(7668.882909864187,4407.159772276878)
best=None
for a,c in zip(l,l[1:]):
    dx,dy=c[0]-a[0],c[1]-a[1];t=max(0,min(1,((A[0]-a[0])*dx+(A[1]-a[1])*dy)/(dx*dx+dy*dy)));q=(a[0]+t*dx,a[1]+t*dy);d=math.hypot(q[0]-A[0],q[1]-A[1])
    if best is None or d<best[0]:best=(d,q,(a,c))
q=best[1];n=2**16
lng=(b['x']+q[0]/4096)/n*360-180;lat=math.degrees(math.atan(math.sinh(math.pi*(1-2*(b['y']+q[1]/4096)/n))))
near=[]
for i,g in enumerate(x['features']):
    if g['layer']=='BldA':
        r=g['geometry']['coordinates'][0] if g['geometry']['type']=='Polygon' else g['geometry']['coordinates'][0][0]
        near.append((round(margin(r,q)*s,1),i))
out['connect']=dict(baseTile=b,bounds=x['bounds'],railFeature=dict(combinedIndex=5948,sourceTile=f['sourceTile'],sourceFeatureIndex=f['sourceFeatureIndex'],properties=f['properties'],coordinates=l),addressPointXY=list(A),footOnRailCenterlineXY=list(q),addressToCenterlineM=round(best[0]*s,2),coordinate=dict(lat=round(lat,7),lng=round(lng,7)),insideAnyBldA=[i for i,g in enumerate(x['features']) if g['layer']=='BldA' and contains_point(g['geometry'],list(q))],nearestBldAM=sorted(near)[:3],metersPerUnit=round(s,4))
json.dump(out,open(W+'candidate-numbers.json','w'),ensure_ascii=False,indent=1)
for k,v in out.items():
    if k=='connect': print(k,v['coordinate'],v['addressToCenterlineM'],v['insideAnyBldA'],v['nearestBldAM']);continue
    print(k,v['alsoContainingOtherBldA'],[(r['combinedIndex'],r['check']['coordinate'],r['check']['mechanicalChecksPassed'],r['approxAreaM2'],r['marginToEdgeM'],r['tileBoundaryTouched'],r['outerBoundaryTouched']) for r in v['features']])
