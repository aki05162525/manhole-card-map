def read(p):return json.loads(Path(p).read_text())
def write(p,o):Path(p).write_text(json.dumps(o,ensure_ascii=False,indent=2)+'\n')
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def canon(o):return hashlib.sha256(json.dumps(o,sort_keys=True,ensure_ascii=False).encode()).hexdigest()
report=read(D/'all-outcomes.json');p=D/'peninsula/hayama-followup/outcome.json';r=read(p)

r['evidenceRefs']=[{'path':x,'sha256':sha(x)} for x in [str(p),r['observationRef'],r['numericRef'],r['userReportRef']]]
report['rows']=[r if old['facilityId']==r['facilityId'] else old for old in report['rows']]
report['priorValidationCommit']='fe6e5f1a945f0562277d604b103aaefda42ca979';write(D/'all-outcomes.json',report)
loc=read(D/'locations-all-proposed.json');fid=r['facilityId'];loc['updates']=[x for x in loc['updates'] if x['facilityId']!=fid]+read(D/'peninsula/hayama-followup/locations-proposed.json')['updates'];loc['updates'].sort(key=lambda x:x['facilityId']);write(D/'locations-all-proposed.json',loc)
lc=read(D/'lifecycle-all-proposed.json');lc['bindings']=[x for x in lc['bindings'] if x['facilityId']!=fid]+read(D/'peninsula/hayama-followup/lifecycle-proposed.json')['bindings'];lc['bindings'].sort(key=lambda x:x['facilityId']);write(D/'lifecycle-all-proposed.json',lc)
sel=read(D/'selection-all-proposed.json');sel['locations']['approved']=[{'facilityId':x['facilityId'],'sha256':canon(x)} for x in loc['updates']];sel['locationLifecycle']['sha256']=sha(D/'lifecycle-all-proposed.json');write(D/'selection-all-proposed.json',sel)
questions=read(D/'human-questions-all.json')
for q in questions['questions']:
 if q['facilityId']==fid:q.update(status='answered-user-report-recompared',answerReceived=True,userReportedAt='2026-11-06',answerRef=r['userReportRef'],resolutionRef=r['observationRef'])
write(D/'human-questions-all.json',questions)
