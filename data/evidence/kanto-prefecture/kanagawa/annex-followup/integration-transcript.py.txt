import json,hashlib,copy
from pathlib import Path
D=Path('data/national/kanto-prefecture/kanagawa'); E=Path('data/evidence/kanto-prefecture/kanagawa'); F=E/'annex-followup'
def read(p):return json.loads(Path(p).read_text())
def write(p,o):Path(p).write_text(json.dumps(o,ensure_ascii=False,indent=2)+'\n')
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def canonical(o):return hashlib.sha256(json.dumps(o,sort_keys=True,ensure_ascii=False).encode()).hexdigest()
rows=[]
for g in ['annex-followup','north','west','peninsula']:
 p=D/g/'outcomes.json';o=read(p)
 for r in o.get('rows',o.get('facilities',[])):
  r=copy.deepcopy(r);obs=read(r['observationRef']);r['performedMethods']=r.get('performedMethods',obs.get('performedMethods',obs.get('performed',[])))
  assert r['performedMethods'],r['facilityId']
  r['stage']='candidate_unapproved' if r['candidateCoordinate'] else 'researched_hold'
  r['adoptionApproved']=False;r['manifestUpdated']=False
  refs=[str(p),r['observationRef']]+([r['numericRef']] if r['numericRef'] else [])
  r['evidenceRefs']=[{'path':x,'sha256':sha(x)} for x in refs];rows.append(r)
rows.sort(key=lambda r:r['facilityId'])
write(D/'all-outcomes.json',{'schema':1,'asOf':'2026-11-05','manifestUpdated':False,'adoptionApproved':False,'rows':rows})
updates=[]
for g in ['', 'north','west','peninsula']:
 updates+=read(D/g/'locations-proposed.json')['updates']
updates.sort(key=lambda r:r['facilityId'])
assert {r['facilityId'] for r in updates}=={r['facilityId'] for r in rows if r['candidateCoordinate']}
write(D/'locations-all-proposed.json',{'schema':1,'updates':updates})
scope={r['currentPlace']['id']:r['currentPlace'] for r in read(D/'scope.json')['facilities']}
bindings=[{'facilityId':r['facilityId'],'identity':{k:scope[r['facilityId']][k] for k in ['name','address']},'adoption':r,'adoptionSHA256':canonical(r),'sourceVersion':'GSI optimal_bvmap PMTiles 2026-07-01; unapproved candidate'} for r in updates]
write(D/'lifecycle-all-proposed.json',{'schema':1,'bindings':bindings,'events':[],'groups':[]})
write(D/'selection-all-proposed.json',{'schema':1,'approvedForIntegration':False,'supersedesForProposedSelection':'selection-proposed.json; select either initial or all, never both','locations':{'path':str(D/'locations-all-proposed.json'),'approved':[{'facilityId':r['facilityId'],'sha256':canonical(r)} for r in updates]},'locationLifecycle':{'path':str(D/'lifecycle-all-proposed.json'),'sha256':sha(D/'lifecycle-all-proposed.json')}})
q=read(D/'human-questions-all.json')
q['questions']=[r for r in q['questions'] if not r['facilityId'].endswith('integration-10')]
for r in q['questions']:
 fid=r['facilityId'];r['answerReceived']=False
 if fid.endswith('0ded97a30e37e98782cb'):
  r.update(status='withdrawn-resolved-independent-source',resolutionRef='data/evidence/kanto-prefecture/kanagawa/peninsula/miura-observation.json')
 elif fid.endswith('ee83b23f6b285cc3df5a'):
  r.update(status='answered-user-report-cardinal-conflict-rechecked',answerReceived=True,userReportedAt='2026-11-06',answerRef=str(F/'user-report-20261106.json'),resolutionRef='data/evidence/kanto-prefecture/kanagawa/west/isehara/observation.json')
 elif fid.endswith('ec19477e52b814868f78'):
  r.update(status='answered-user-report-independent-footprint-unresolved',answerReceived=True,userReportedAt='2026-11-06',answerRef=str(F/'user-report-20261106.json'))
 else:r['status']='sent-to-parent-after-independent-research'
for r in rows:
 if r['humanQuestion'] and r['facilityId'].endswith('integration-10'):
  q['questions'].append({'facilityId':r['facilityId'],'name':r['name'],**r['humanQuestion'],'status':'sent-to-parent-after-independent-research','answerReceived':False,'noMunicipalContact':True})
write(D/'human-questions-all.json',q)
print(len(rows),'researched',len(updates),'unapproved candidates')
