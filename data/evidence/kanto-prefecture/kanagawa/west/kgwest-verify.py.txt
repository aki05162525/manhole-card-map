from location_lifecycle import digest
E=Path('data/evidence/kanto-prefecture/kanagawa/west');N=Path('data/national/kanto-prefecture/kanagawa/west')
inputs={};count=0
for r in json.load(open(E/'retrievals.json'))['records']:
 if r.get('sha256') and r.get('rawPath'):
  p=Path(r['rawPath']);assert hashlib.sha256(p.read_bytes()).hexdigest()==r['sha256'],p;inputs[str(p)]=r['sha256'];count+=1

 if not row['numericRef']:assert row['candidateCoordinate'] is None;continue
 p=Path(row['numericRef']);v=json.load(open(p));tiles=[]
 for r in json.load(open(p.parent/'tile-inputs.json'))['rows']:
  raw=Path(r['path']).read_bytes();assert hashlib.sha256(raw).hexdigest()==r['sha256'];inputs[r['path']]=r['sha256'];d=decode_tile(raw);features=[]

   for i,f in enumerate(d.get(layer,{}).get('features',[])):features.append({**f,'id':f"{r['tile']['x']}-{r['tile']['y']}-{layer}-{i}",'sourceLayer':layer,'layerFeatureIndex':i})
  tiles.append({'tile':r['tile'],'extent':4096,'features':features})
 c=combine_tiles(tiles);f=next(f for f in c['features'] if f['id']==v['feature']['id']);assert f==v['feature'];assert shape(f['geometry']).is_valid
 r=rounded_coordinate(f['geometry'],c['baseTile'],v['pointXY']);assert r['coordinate']==v['coordinate']==row['candidateCoordinate'];assert r['unroundedWithin'] and r['roundedWithin'];assert not f['tileBoundaryTouched'];assert not f['outerBoundaryTouched'];print(row['facilityId'],f['id'],r['coordinate'],'valid / round-trip inside / unclipped')

for x,y,z in zip(a,b,sel['locations']['approved']):
 assert x==y['adoption'];assert digest(x)==y['adoptionSHA256']==z['sha256'];assert hashlib.sha256(Path(x['evidence']['snapshotPath']).read_bytes()).hexdigest()==x['evidence']['snapshotSHA256']
assert hashlib.sha256((N/'lifecycle-proposed.json').read_bytes()).hexdigest()==sel['locationLifecycle']['sha256']
for p in list(E.rglob('*.json'))+list(N.rglob('*.json')):
 if p!=E/'verification.json':inputs[str(p)]=hashlib.sha256(p.read_bytes()).hexdigest()
