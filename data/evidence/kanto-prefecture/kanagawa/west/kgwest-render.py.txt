import json,re,sys
from pathlib import Path
import cairosvg
from shapely.geometry import shape
root=Path.cwd();sys.path.insert(0,str(root/'tools/national'));from research_common import render_svg
name=sys.argv[1];W=root/'data/work/kanto-kanagawa/west'/name;E=root/'data/evidence/kanto-prefecture/kanagawa/west'/name
c=json.loads((W/'combined.json').read_text());s=render_svg(c)
if len(sys.argv)>2:
 bounds=[float(v) for v in sys.argv[2:]]
 s=re.sub(r'viewBox="[^"]+"','viewBox="'+' '.join(sys.argv[2:])+'"',s,count=1)
else:bounds=c['bounds'];bounds=[bounds[0],bounds[1],bounds[2]-bounds[0],bounds[3]-bounds[1]]
x,y,w,h=bounds;labels=[];rows=[]
for f in c['features']:
 if f['sourceLayer']!='BldA':continue
 g=shape(f['geometry']);p=g.representative_point()
 if x<p.x<x+w and y<p.y<y+h and g.area>1500:
  lab=f'{f["sourceTile"]["x"]-c["baseTile"]["x"]},{f["sourceTile"]["y"]-c["baseTile"]["y"]}:{f["layerFeatureIndex"]}'
  labels.append(f'<text x="{p.x}" y="{p.y}" font-size="{w/90}" fill="blue">{lab}</text>');rows.append({'label':lab,'id':f['id'],'bounds':g.bounds,'area':g.area})
s=s.replace('</svg>','\n'.join(labels)+'</svg>');(E/'detail.svg').write_text(s);cairosvg.svg2png(bytestring=s.encode(),write_to=str(W/'detail.png'),output_width=1800,output_height=1800,background_color='white');(W/'labels.json').write_text(json.dumps(rows))
