function initMap() { var latlng = { lat: 33.561242, lng: 133.572944 }, map = new google.maps.Map(document.getElementById('googleMaps'), { center: latlng, zoom: 14, disableDefaultUI: true, styles: [{ featureType: 'all', elementType: 'all', stylers: [{ hue: '#B1C4E6' }, { saturation: -50 }, { lightness: 0 }, { gamma: 1 }] }] }), marker = new google.maps.Marker({ position: latlng, map: map, icon: '/share/img/top/icon_map.png' }); google.maps.event.addDomListener( window, 'load', initMap ); }
